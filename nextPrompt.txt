  /// Box expressed in normalized capture-image coordinates (0..1), centered.
  final double widthRatio;   // e.g. 0.7 of image width
  final double heightRatio;  // e.g. 0.7 of image height
  const PlacementBox({this.widthRatio = 0.7, this.heightRatio = 0.7});

  Rect get normalizedRect => Rect.fromCenter(

    width: widthRatio,
    height: heightRatio,

}

### Step 1: Smoothed pitch provider

```dart
class PitchSample {
  final double pitchDegrees;     // smoothed
  final bool sensorSupported;
  const PitchSample(this.pitchDegrees, this.sensorSupported);
}

final currentPitchProvider = StreamProvider<PitchSample>((ref) {
  final channel = SensorStreamChannel();
  double? smoothed;
  const alpha = 0.15; // low-pass factor; tune for responsiveness vs stability
  return channel.stream.map((p) {
    if (!p.deviceMotionSupported) return const PitchSample(0, false);
    final raw = p.beta; // pitch in degrees per the agreed convention — CONFIRM field/sign
    smoothed = (smoothed == null) ? raw : (alpha * raw + (1 - alpha) * smoothed!);
    return PitchSample(smoothed!, true);
  });
});
```

  final Stability stability;
  final bool sensorSupported;
  const StabilitySample(this.stability, this.sensorSupported);

final stabilityProvider = StreamProvider<StabilitySample>((ref) {
  final channel = SensorStreamChannel();
  double emaMagnitude = 0;
  const alpha = 0.2;                // smoothing
  // Stability decision uses hysteresis + a time hold (see Step 2).
  return channel.stream.map((p) {
    if (!p.deviceMotionSupported) return const StabilitySample(Stability.unknown, false);
    // user acceleration magnitude (g). If rotation rate is available, fold it in.
    final acc = p.accelerometer;
    final mag = sqrt(acc.x * acc.x + acc.y * acc.y + acc.z * acc.z);
    emaMagnitude = alpha * mag + (1 - alpha) * emaMagnitude;

    return StabilitySample(s, true);
  });
});
